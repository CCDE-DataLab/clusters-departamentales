\vspace{.06cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{.485\textwidth}
\begin{keybox}[equal height group=g3,top=1.5mm,bottom=1.5mm]
\termino{Valor Agregado Bruto (VAB real)}{Medida del INEI a precios constantes de 2007 en \Departments\ regiones y \Sectors\ ramas (componente del PBI antes de impuestos netos).}
\termino{Índice de volumen ($I_{it}$, 2019 = 100)}{Compara el VAB real de cada región con su nivel de 2019; \EnclaveFinal\ indica un volumen 2.4\% menor al prepandemia.}
{\small\textcolor{ccdeaccent}{\bfseries Tasa anual ($g_{it}$) vs. nivel acumulado ($I_{it}$)}}\par{\footnotesize La tasa mide el cambio porcentual anual; el índice evalúa si se recuperó el volumen previo al choque.}
\end{keybox}
\end{column}
\begin{column}{.485\textwidth}
\begin{keybox}[equal height group=g3,top=1.5mm,bottom=1.5mm]
\termino{¿Por qué usar coordenadas CLR / ILR?}{Las \Sectors\ ramas suman 100\% (símplex): aumentar una participación obliga a reducir otra y sesga la distancia euclidiana directa.}
\termino{Transformación de Aitchison}{$\operatorname{clr}(x_{ij})=\ln x_{ij}-\overline{\ln x_i}$ centra cada rama respecto a la media geométrica regional (\PreStart--\PreEnd, \Zeros\ ceros) y la proyecta a $\mathbb{R}^D$.}
{\small\textcolor{ccdeaccent}{\bfseries Equivalencia isométrica (ILR)}}\par{\footnotesize Con base de Helmert en $\mathbb{R}^{10}$, la distancia euclidiana en CLR/ILR coincide con la distancia de Aitchison.}
\end{keybox}
\end{column}
\end{columns}
\end{frame}

% ---------------------------------------------------------------- 3. Metodología de clustering y perfiles
\begin{frame}[t]{Construcción de la tipología propia: de CLR y PCA a los cuatro perfiles Ward}
\vspace{.06cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{.46\textwidth}
\begin{keybox}[equal height group=g4,top=1.6mm,bottom=1.6mm]
\textbf{Algoritmo y espacio de agrupamiento}\par\vspace{.05cm}
\footnotesize
\textbf{1. Reducción PCA:} sobre CLR centrado sin estandarizar; 2 ejes retienen \textbf{\PcaCumulative\%} de la varianza (PC1 \PcaOne\% extractivo; PC2 \PcaTwo\% manufactura vs. agro).\par\vspace{.05cm}
\textbf{2. Cúmulos jerárquicos de Ward:} minimiza la varianza intragrupo ($\Delta\text{WCSS}$) en cada fusión; coincide al 100\% con K-Means en $K=4$ sobre PCA2.\par\vspace{.05cm}
\textbf{3. Contraste de \SpecCount\ variantes:} 4 métodos (Ward, K-Means, PAM, GMM) $\times$ 3 espacios (PCA2, PCA3, ILR) $\times$ $K\in\{2..5\}$, validados con Silueta, Gap y Jaccard sin reemplazo.
\end{keybox}
\end{column}
\begin{column}{.51\textwidth}
\begin{keybox}[equal height group=g4,top=1.6mm,bottom=1.6mm]
\textbf{Los 4 perfiles generados ($K=4$, ordenados por minería)}\par\vspace{.05cm}
\scriptsize
\textbf{1. Diversificado/Agro ($n=\AgroCount$):} 0.9\% minería, 12.7\% manufactura (\emph{Lima, Lambayeque, San Martín}). Grupo base.\par\vspace{.04cm}
\textbf{2. Servicios/Manufactura ($n=\ServicesCount$):} 12.2\% minería, 10.6\% manuf. (\emph{Ica, La Libertad, Piura, Puno, Cajamarca y 6 más}).\par\vspace{.04cm}
\textbf{3. Minero/Diversificado ($n=\MiningCount$):} 37.4\% minería, \textbf{12.2\% manuf.} (\emph{Arequipa, Áncash, Cusco, Junín, Moquegua, Tacna, Madre de Dios}).\par\vspace{.04cm}
\textbf{4. Minero/Enclave ($n=\EnclaveCount$):} 41.9\% minería, \textbf{1.8\% manuf.} (\emph{Apurímac, Huancavelica, Pasco}).\par\vspace{.05cm}
\textcolor{ccdeaccent}{\bfseries Nota:} etiquetas analíticas propias; no son categorías oficiales del INEI.
\end{keybox}
\end{column}
\end{columns}
\end{frame}

% ---------------------------------------------------------------- 13. Límites
\begin{frame}[t]{Limitaciones del análisis: tamaño muestral, choques simultáneos y sensibilidad}
\vspace{.06cm}
\begin{keybox}
\textbf{1. Tamaño muestral del enclave ($N=\EnclaveCount$).} Incluye Apurímac, Pasco y Huancavelica; esta última ingresa por su baja manufactura (1.7\%), pese a tener minería moderada (13.4\%).\par\smallskip
\textbf{2. Simultaneidad de choques.} El periodo \PostStart--\PostEnd\ coincide con bloqueos en el Corredor Minero y menor ley en tajos maduros de Pasco, factores no separables con series anuales hasta \LevelEnd.\par\smallskip
\textbf{3. Sensibilidad dimensional.} Con 3 ejes PCA la concordancia baja (ARI = 0.57); con coordenadas ILR completas se mantiene alta (ARI = 0.87).
\end{keybox}
\vspace{.08cm}
\begin{simplebox}
\nota{Alcance empírico} documenta una divergencia robusta en el nivel recuperado, pero requiere datos de mayor frecuencia para identificar causas.
\end{simplebox}
\end{frame}

\begin{frame}[t]{Conclusiones: respuesta a la pregunta empírica sobre recuperación regional}
\begin{columns}[T,onlytextwidth]
\begin{column}{.485\textwidth}
\begin{keybox}[equal height group=g15,top=1.5mm,bottom=1.5mm]
\termino{1. Articulación y recuperación observada}{Los perfiles con mayor articulación sectorial muestran mejor recuperación observada: en 2023 la minería diversificada superó a 2019 (\MiningFinal) y los enclaves promediaron \EnclaveFinal.}
\termino{2. Crecer tras caer no es recuperarse}{Las tasas positivas de 2021--2023 no compensaron la caída inicial; en nivel ($2019=100$), los enclaves registran un cambio relativo de \textbf{\ScarBeta\ puntos} frente a diversificado/agro ($p=\ScarP$).}
\end{keybox}
\end{column}
\begin{column}{.485\textwidth}
\begin{keybox}[equal height group=g15,top=1.5mm,bottom=1.5mm]
\termino{3. Dispersión concentrada en un episodio}{La mayor volatilidad del grupo enclave en tasas anuales (\SdEnclave\ pp) responde en gran parte a la observación extrema de Apurímac en 2016 (+\ExtremeGrowth\%); sin ella, baja a 7.0 pp.}
\termino{4. Alcance descriptivo y monitoreo}{La evidencia documenta heterogeneidad y persistencia del rezago en niveles sin atribuir causalidad única; el seguimiento regional debe combinar tasas anuales e índices de volumen.}
\end{keybox}
\end{column}

\begin{frame}[t]{Anexo · Especificación técnica de los ajustes metodológicos}
\vspace{.08cm}
{\footnotesize
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{keybox}[equal height group=g16]
\textbf{Transformación composicional (CLR / ILR).} Las participaciones pertenecen al símplex ($\sum x_j = 100\%$). La transformación log-cociente centrada (Aitchison, 1986) proyecta los datos al espacio real antes de medir distancias.\par\smallskip
\textbf{Tratamiento de ceros.} Reemplazo multiplicativo ($\delta=\DeltaZero$, Martín-Fernández et al., 2003); los promedios utilizados registran \Zeros\ ceros, por lo que no altera la composición en $\delta \in [0.01, 0.10]$.\par\smallskip
\textbf{Reducción y agrupamiento.} PCA retiene \PcaCumulative\% en 2 ejes; Ward minimiza la varianza intragrupo.
\end{keybox}
\end{column}
\begin{column}{.48\textwidth}
\begin{keybox}[equal height group=g16]
\textbf{Validación de cúmulos.} Contraste conjunto de silueta, estadístico Gap (Tibshirani et al., 2001) y estabilidad de Jaccard por submuestreo sin reemplazo (Hennig, 2007).\par\smallskip
\textbf{Efectos fijos de entidad.} Controlan por heterogeneidad departamental inobservable e invariante en el tiempo.\par\smallskip
\textbf{Inferencia agrupada (Cluster SE).} Corrige la correlación serial dentro de cada departamento.\par\smallskip
\textbf{Sensibilidad.} M4 evalúa la influencia de Apurímac 2016; M6 usa intensidad minera continua.
\end{keybox}
\end{column}
\end{columns}
}
\end{frame}
